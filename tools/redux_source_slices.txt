012824: function TAP_runGraXpertCLI( args, expectedOutput )
012825: {
012826:    var candidates = TAP_graxpertCandidateExecutables();
012827:    var lastError = "";
012828: 
012829:    for ( var i = 0; i < candidates.length; ++i )
012830:    {
012831:       var exe = candidates[i];
012832: 
012833:       // Skip missing absolute/local paths, but allow PATH command names.
012834:       if ( ( exe.indexOf( "/" ) >= 0 || exe.indexOf( "\\" ) >= 0 ) &&
012835:            !_cc_fileExistsSafe( exe ) )
012836:          continue;
012837: 
012838:       console.noteln( "Trying GraXpert launcher: " + exe );
012839: 
012840:       var r = runExternalProcessBlocking(
012841:          exe, args,
012842:          {
012843:             onStdOutText: function( t )
012844:             {
012845:                var lines = String( t ).split( /\r?\n/ );
012846:                for ( var k = 0; k < lines.length; ++k )
012847:                   if ( lines[k].length > 0 )
012848:                      console.writeln( "GraXpert: " + lines[k] );
012849:             },
012850:             onStdErrText: function( t )
012851:             {
012852:                var lines = String( t ).split( /\r?\n/ );
012853:                for ( var k = 0; k < lines.length; ++k )
012854:                   if ( lines[k].length > 0 )
012855:                      console.warningln( "GraXpert: " + lines[k] );
012856:             }
012857:          }
012858:       );
012859: 
012860:       if ( r.started )
012861:       {
012862:          if ( exe.indexOf( "/" ) >= 0 || exe.indexOf( "\\" ) >= 0 )
012863:             TAP_saveGraXpertExecutable( exe );
012864: 
012865:          if ( _cc_waitForFileReady( expectedOutput, 300000, 1000 ) )
012866:             return;
012867: 
012868:          lastError = r.stderr && r.stderr.length > 0
012869:             ? r.stderr
012870:             : "GraXpert started but did not generate the expected output.";
012871:          break;
012872:       }
012873:    }
012874: 
012875:    // No candidate started: ask once, without adding another permanent UI field.
012876:    if ( lastError.length == 0 )
012877:    {
012878:       var chosen = TAP_promptForGraXpertExecutable();
012879:       if ( chosen.length == 0 )
012880:          throw new Error(
012881:             "GraXpert executable was not found automatically. " +
012882:             "Install GraXpert or select its Windows executable when prompted."
012883:          );
012884: 
012885:       var rr = runExternalProcessBlocking( chosen, args, {} );
012886:       if ( rr.started && _cc_waitForFileReady( expectedOutput, 300000, 1000 ) )
012887:          return;
012888: 
012889:       lastError = rr.stderr && rr.stderr.length > 0
012890:          ? rr.stderr
012891:          : "Selected GraXpert executable did not produce the expected output.";
012892:    }
012893: 
012894:    throw new Error( "GraXpert execution failed.\n" + lastError );
012895: }

028630: function TAP_executeWorkflowStage( stageId, ctx, s )
028631: {
028632:    var workView = ctx.workView;
028633: 
028634:    switch ( stageId )
028635:    {
028636:    case "optical":
028637:       if ( s.opticalEngine == ENGINE_SASPRO )
028638:          executeSASproCorrectOnly( workView, s );
028639:       else
028640:          executeBlurXCorrectOnly( workView, s );
028641: 
028642:       if ( s.autoSTF ) applyAutoSTF( workView, false );
028643:       createProcessSnapshotIfEnabled(
028644:          workView, s, s.opticalEngine == ENGINE_SASPRO ? "SASpro_correctonly" : "correctonly"
028645:       );
028646:       break;
028647: 
028648:    case "adbe":
028649:       var hadAstrometryBeforeADBE = TAP_hasAstrometricSolution( workView );
028650: 
028651:       if ( s.adbeEngine == ADBE_ENGINE_GRAXPERT )
028652:          executeGraXpertAutoDBE( workView, s.adbeDivideFirst, s.graxpertSmoothing );
028653:       else
028654:          executeSetiAstroAutoDBE( workView, s.adbeDivideFirst );
028655: 
028656:       if ( hadAstrometryBeforeADBE && !TAP_hasAstrometricSolution( workView ) )
028657:          throw new Error(
028658:             "The image lost its astrometric solution during Automatic DBE. " +
028659:             "The workflow cannot continue safely to SPCC."
028660:          );
028661:       if ( s.autoSTF ) applyAutoSTF( workView, false );
028662:       createProcessSnapshotIfEnabled( workView, s, s.adbeEngine == ADBE_ENGINE_GRAXPERT ? "GraXpert" : "DBE" );
028663:       break;
028664: 
028665:    case "spcc":
028666:       var spccCompleted = executeSPCC( workView, s );
028667:       if ( spccCompleted )
028668:       {
028669:          if ( s.autoSTF ) applyAutoSTF( workView, true );
028670:          createProcessSnapshotIfEnabled( workView, s, "SPCC" );
028671:       }
028672:       else
028673:          console.warningln( "SPCC stage skipped. Continuing with the remaining selected stages." );
028674:       break;
028675: 
028676:    case "blurx":
028677:       if ( s.sharpenEngine == ENGINE_SASPRO )
028678:          executeSASproSharpen( workView, s );
028679:       else
028680:          executeBlurXSharpen( workView, s );
028681: 
028682:       if ( s.autoSTF ) applyAutoSTF( workView, true );
028683:       createProcessSnapshotIfEnabled(
028684:          workView, s, s.sharpenEngine == ENGINE_SASPRO ? "SASpro_Sharpen" : "BlurX"
028685:       );
028686:       break;
028687: 
028688:    case "noisex":
028689:       if ( s.noiseEngine == ENGINE_SASPRO )
028690:          executeSASproDenoise( workView, s );
028691:       else
028692:          executeNoiseX( workView, s );
028693: 
028694:       createProcessSnapshotIfEnabled(
028695:          workView, s, s.noiseEngine == ENGINE_SASPRO ? "SASpro_Denoise" : "NoiseX"
028696:       );
028697:       break;
028698: 
028699:    case "starx":
028700:       ctx.idsBeforeStarX = TAPSTAR_getAllImageIDs();
028701: 
028702:       if ( s.starRemovalEngine == STAR_ENGINE_STARNET )
028703:       {
028704:          if ( s.starStretch && !s.starnetMask )
028705:             throw new Error(
028706:                "Star Stretch is enabled, but StarNet2 > Mask is disabled. " +
028707:                "Enable Mask so the workflow can retain an extracted stars image."
028708:             );
028709: 
028710:          ctx.starsView = executeStarNet2( workView, s );
028711:          if ( s.starnetMask && ctx.starsView == null )
028712:             throw new Error(
028713:                "StarNet2 completed, but its generated star mask/output image could not be located."
028714:             );
028715:          if ( ctx.starsView != null )
028716:             console.writeln( "Resolved stars image : " + ctx.starsView.id );
028717: 
028718:          TAP_collectGarbage( "Star Removal stage complete" );
028719:          if ( s.backupEachProcess )
028720:             console.warningln(
028721:                "Creating the post-StarNet2 backup will duplicate the full starless image in memory. " +
028722:                "If PixInsight is memory constrained, disable 'Create a copy of the file after every process'."
028723:             );
028724:          createProcessSnapshotIfEnabled( workView, s, "StarNet2" );
028725:       }
028726:       else
028727:       {
028728:          if ( s.starStretch && !s.starOutputStars )
028729:             throw new Error(
028730:                "Star Stretch is enabled, but StarXTerminator > Output Stars is disabled."
028731:             );
028732: 
028733:          executeStarX( workView, s );
028734:          if ( s.starOutputStars )
028735:          {
028736:             ctx.starsView = TAPSTAR_resolveStarsView( workView, ctx.idsBeforeStarX );
028737:             if ( ctx.starsView == null )
028738:                throw new Error(
028739:                   "StarXTerminator completed, but the generated stars image could not be located. " +
028740:                   "Processing has been stopped before the next stage to avoid using an invalid view."
028741:                );
028742:             console.writeln( "Resolved stars image : " + ctx.starsView.id );
028743:          }
028744: 
028745:          // Star removal can temporarily consume substantial CPU/GPU/RAM resources.
028746:          // Clean up the native process wrapper before an optional full-resolution backup.
028747:          TAP_collectGarbage( "Star Removal stage complete" );
028748:          if ( s.backupEachProcess )
028749:             console.warningln(
028750:                "Creating the post-StarX backup will duplicate the full starless image in memory. " +
028751:                "If PixInsight is memory constrained, disable 'Create a copy of the file after every process'."
028752:             );
028753:          createProcessSnapshotIfEnabled( workView, s, "StarX" );
028754:       }
028755:       break;

028766:          );

028769:          workView,
028770:          ctx.backgroundResult.rect
028771:       );
028772: 
028773:       createProcessSnapshotIfEnabled( workView, s, "BG_Neutralization" );
028774:       break;
028775: 
028776:    case "statstretch":
028777:       executeImageStretch( workView, s );
028778:       ctx.nonlinearReady = true;
028779:       createProcessSnapshotIfEnabled( workView, s, "stretch" );
028780:       break;
